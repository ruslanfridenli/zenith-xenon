d = [12.5, 7.0, 23.75, 4.5, 18.0, 9.25, 31.0]
sum = Enum.sum(d)
IO.puts("count : #{length(d)}")
IO.puts(:io.format("sum   : ~.2f~n", [sum]))
IO.puts(:io.format("mean  : ~.2f~n", [sum / length(d)]))
IO.puts(:io.format("min   : ~.2f~n", [Enum.min(d)]))
IO.puts(:io.format("max   : ~.2f~n", [Enum.max(d)]))
